\sectitle{1}{Part A --- Written Problems (25 pts)}

\qtitle{A1 --- Forward pass by hand}{[4 pts]}
\begin{answerbox}
Given a two-layer network with weights
\[
\vect W_1=\begin{pmatrix}1&-1\\0.5&1\end{pmatrix},\qquad
\vect b_1=\begin{pmatrix}0\\-0.5\end{pmatrix},\qquad
\vect x=\begin{pmatrix}1\\2\end{pmatrix},
\]
\[
\vect W_2=\begin{pmatrix}1&0\\-1&2\end{pmatrix},\qquad
\vect b_2=0,
\]
and true class $y_0=1$ (one-hot $\vect y=(1,0)$):

\textbf{(a) Hidden layer.} With $\vect z_1=\vect W_1\vect x+\vect b_1$,
\[
\begin{aligned}
\vect z_1
&=\begin{pmatrix}1&-1\\0.5&1\end{pmatrix}\begin{pmatrix}1\\2\end{pmatrix}
 +\begin{pmatrix}0\\-0.5\end{pmatrix} \\
&=\begin{pmatrix}1(1)+(-1)(2)\\0.5(1)+1(2)\end{pmatrix}
 +\begin{pmatrix}0\\-0.5\end{pmatrix}
 =\begin{pmatrix}-1\\2.5\end{pmatrix}+\begin{pmatrix}0\\-0.5\end{pmatrix}
 =\begin{pmatrix}-1\\2\end{pmatrix}.
\end{aligned}
\]
\[
\vect h=\operatorname{ReLU}(\vect z_1)=\bigl(\max(0,-1),\max(0,2)\bigr)=(0,2).
\]

\textbf{(b) Output layer.} With $\vect z_2=\vect W_2\vect h+\vect b_2$,
\[
\vect z_2=\begin{pmatrix}1&0\\-1&2\end{pmatrix}\begin{pmatrix}0\\2\end{pmatrix}+0
=\begin{pmatrix}1(0)+0(2)\\-1(0)+2(2)\end{pmatrix}
=\begin{pmatrix}0\\4\end{pmatrix}.
\]
\[
\vect p=\operatorname{softmax}(\vect z_2)
=\left(\frac{e^0}{e^0+e^4},\frac{e^4}{e^0+e^4}\right)
=\left(\frac{1}{1+e^4},\frac{e^4}{1+e^4}\right)
\approx(0.01799,0.98201).
\]

\textbf{(c) Loss.} Since $\vect y=(1,0)$, only the $k=0$ term of the cross-entropy sum survives:
\[
L=-\sum_k y_k\log p_k=-\log p_0=\log(1+e^4)\approx4.0181.
\]
This large loss is expected: the network's logits strongly favour class 1 ($z_{2,1}=4\gg z_{2,0}=0$), but the true label is class 0, so the model is confidently wrong on this particular hand-picked example.
\end{answerbox}

\qtitle{A2 --- Backprop over the computational graph}{[6 pts]}
\begin{answerbox}
\textbf{(a) The graph.} Every op in the forward pass becomes one node; every tensor that flows between ops becomes a labelled edge. For the network in A1 this is exactly the seven-op chain the implementation uses (\code{src/autograd/tensor.py} for the primitives, \code{src/layers.py} for Linear, \code{src/losses.py} for the fused softmax + cross-entropy).

\vspace{1.5mm}
\begin{center}
\includegraphics[width=\linewidth]{assets/a2_mlp_computational_graph.png}
\end{center}
\vspace{1mm}

\textbf{Figure 1:} Task-specific two-layer MLP computational graph. The diagram follows the complete forward path from the two-dimensional input $\vect x$, through the first affine transformation $\vect z_1=\vect W_1\vect x+\vect b_1$, ReLU activation $\vect h$, the second affine transformation $\vect z_2=\vect W_2\vect h+\vect b_2$, softmax probabilities $\vect p$, and the one-hot target $\vect y$ into the cross-entropy loss $L$. Each stage is separated and labelled so that the architecture and tensor flow are readable without introducing an unnecessarily dense generic neural-network drawing.

\textbf{Backward local derivatives, edge by edge} (this is literally what each op's backward closure in \code{tensor.py} computes, right-to-left):
\begin{itemize}
\item \textbf{Cross-Entropy} ($\vect p\to L$): $\partial L/\partial p_k=-y_k/p_k$.
\item \textbf{Softmax} ($\vect z_2\to\vect p$): local Jacobian $J_{ij}=\partial p_i/\partial z_{2,j}=p_i(\delta_{ij}-p_j)$, i.e. $\vect J=\operatorname{diag}(\vect p)-\vect p\vect p^\top$ (derived below).
\item \textbf{Add bias} ($\vect a_2\to\vect z_2$): $\partial\vect z_2/\partial\vect a_2=\vect I$, $\partial\vect z_2/\partial\vect b_2=1$ (summed over the batch axis --- this is exactly \code{Tensor._unbroadcast} undoing the broadcast of $\vect b_2$ over the batch).
\item \textbf{MatMul} ($\vect h\to\vect a_2$): $\partial\vect a_2/\partial\vect h=\vect W_2^\top$, $\partial\vect a_2/\partial\vect W_2=\vect h^\top$ (outer product).
\item \textbf{ReLU} ($\vect z_1\to\vect h$): local derivative is the elementwise mask $\mathbf{1}[\vect z_1>0]$ --- in code, \code{(self.data > 0) * out.grad}.
\item \textbf{Add bias} ($\vect a_1\to\vect z_1$): identity, same as above.
\item \textbf{MatMul} ($\vect x\to\vect a_1$): $\partial\vect a_1/\partial\vect W_1=\vect x^\top$.
\end{itemize}

\textbf{(b) Softmax-cross-entropy gradient, derived (not asserted).} Write $L=-\sum_k y_k\log p_k$ with $p_k=e^{z_k}/\sum_j e^{z_j}$ (dropping the subscript ``2'' for brevity). By the chain rule,
\[
\frac{\partial L}{\partial z_i}
=\sum_k\frac{\partial L}{\partial p_k}\frac{\partial p_k}{\partial z_i}
=\sum_k\left(-\frac{y_k}{p_k}\right)p_k(\delta_{ki}-p_i).
\]
The softmax Jacobian entry $\partial p_k/\partial z_i=p_k(\delta_{ki}-p_i)$ splits into a diagonal case ($k=i$: $\partial p_i/\partial z_i=p_i(1-p_i)$) and an off-diagonal case ($k\ne i$: $\partial p_k/\partial z_i=-p_kp_i$) --- both captured by the single formula above via the Kronecker delta. Substituting and cancelling the $p_k$ in the numerator against the $1/p_k$ from the log-derivative,
\[
\frac{\partial L}{\partial z_i}=\sum_k -y_k(\delta_{ki}-p_i)
=-y_i+p_i\sum_k y_k=p_i-y_i,
\]
using $\sum_k y_k=1$ since $\vect y$ is one-hot. So $\partial L/\partial\vect z_2=\vect p-\vect y$ --- this is exactly why \code{losses.py} is allowed to special-case the backward of \code{softmax_cross_entropy} into one fused line, \code{(p - one_hot) / n}, instead of building softmax and cross-entropy as two separate autograd nodes: the Jacobian-times-gradient product above collapses to a single elementwise subtraction, so computing the full $10\times10$ softmax Jacobian at runtime would be pure waste. Numerically here: $\partial L/\partial\vect z_2=\vect p-\vect y\approx(-0.98201,0.98201)$.

Since $\vect z_2=\vect W_2\vect h+\vect b_2$, the local derivatives are $\partial\vect z_2/\partial\vect W_2=\vect h^\top$ (outer product) and $\partial\vect z_2/\partial\vect b_2=1$, so
\[
\frac{\partial L}{\partial\vect W_2}=(\vect p-\vect y)\vect h^\top
=\begin{pmatrix}0&-1.9640\\0&1.9640\end{pmatrix},\qquad
\frac{\partial L}{\partial\vect b_2}=\vect p-\vect y\approx(-0.98201,0.98201).
\]

\textbf{(c) One layer down.} $\partial L/\partial\vect h=\vect W_2^\top(\vect p-\vect y)\approx(-1.9640,1.9640)$. The ReLU gate multiplies elementwise by $\mathbf{1}[\vect z_1>0]=(0,1)$ (recall $\vect z_1=(-1,2)$), so
\[
\frac{\partial L}{\partial\vect z_1}=(-1.9640,1.9640)\odot(0,1)=(0,1.9640).
\]
Then $\partial L/\partial\vect W_1=(\partial L/\partial\vect z_1)\vect x^\top$:
\[
\frac{\partial L}{\partial\vect W_1}=\begin{pmatrix}0&0\\1.9640&3.9281\end{pmatrix}.
\]
Note that the whole first row is exactly zero --- the first hidden unit's pre-activation was $z_{1,0}=-1<0$, so ReLU zeroed its gradient completely. This is a one-example preview of the dying-ReLU mechanism formalised in A3.
\end{answerbox}

\qtitle{A3 --- Activation derivatives \& the dying-ReLU problem}{[4 pts]}
\begin{answerbox}
\textbf{(a) Sigmoid.} $\sigma(z)=\frac{1}{1+e^{-z}}=(1+e^{-z})^{-1}$. By the chain rule,
\[
\sigma'(z)=-(1+e^{-z})^{-2}\cdot(-e^{-z})
=\frac{e^{-z}}{(1+e^{-z})^2}
=\frac{1}{1+e^{-z}}\cdot\frac{e^{-z}}{1+e^{-z}}
=\sigma(z)\bigl(1-\sigma(z)\bigr),
\]
using $1-\sigma(z)=\frac{e^{-z}}{1+e^{-z}}$. Since $\sigma(z)\in(0,1)$, the product $\sigma(1-\sigma)$ is a downward parabola in $\sigma$ maximised at $\sigma=0.5$ (i.e. $z=0$), giving $\sigma'(0)=0.25$ --- the single largest gradient a sigmoid unit can ever pass backward.

\textbf{Tanh.} From $\tanh(z)=\frac{e^z-e^{-z}}{e^z+e^{-z}}$, the quotient rule gives
\[
\tanh'(z)=\frac{(e^z+e^{-z})^2-(e^z-e^{-z})^2}{(e^z+e^{-z})^2}
=1-\tanh^2(z),
\]
since the numerator is $(e^z+e^{-z})^2-(e^z-e^{-z})^2$ and $\tanh^2(z)=(e^z-e^{-z})^2/(e^z+e^{-z})^2$. The implementation (\code{src/activations.py}) does not use this exp-based formula directly; it instead builds tanh as $2\sigma(2z)-1$, which is algebraically identical (substituting $\sigma(2z)$ and simplifying recovers the same $\tanh(z)$) but reuses the already-stabilised sigmoid --- see the numerical-stability note below.

\textbf{ReLU.} $\operatorname{ReLU}(z)=\max(0,z)$, so $\operatorname{ReLU}'(z)=1$ if $z>0$, $0$ if $z<0$ (undefined at $z=0$; the implementation conventionally takes 0 there, since \code{(self.data > 0)} is False at exactly 0).

\textbf{(b) The dying-ReLU mechanism.} A ReLU unit is defined by a pre-activation $z=\vect w^\top\vect x+b$ computed before the nonlinearity. If a large update (e.g. from a too-high learning rate or an outlier gradient) pushes $\vect w,b$ into a region where $z\le0$ for every training example, then $\operatorname{ReLU}'(z)=0$ for all of them, so by the chain rule the gradient of the loss with respect to that unit's incoming weights is exactly zero: $\partial L/\partial\vect w=(\partial L/\partial\operatorname{ReLU}(z))\cdot0=0$, regardless of how large the upstream error signal $\partial L/\partial\operatorname{ReLU}(z)$ is. Since the gradient is zero, the optimizer's update to $\vect w,b$ is also zero, so $z$ never moves back above 0 on its own --- the unit is permanently stuck outputting 0 and contributing nothing to the forward pass either. This is exactly the mechanism \code{test_dying_relu_zero_gradient} in \code{tests/test_components.py} checks directly: feeding an all-negative pre-activation through \code{Tensor.relu()} and confirming the backward pass leaves \code{pre_act.grad} identically zero. One standard fix is Leaky ReLU (or ELU/PReLU), which keeps a small nonzero slope for $z<0$ so a dead unit still receives some gradient and can recover.
\end{answerbox}

\qtitle{A4 --- The loss is a negative log-likelihood}{[3 pts]}
\begin{answerbox}
\textbf{(a)} For a one-hot label $\vect y\sim\operatorname{Cat}(\vect p)$, the likelihood of observing the class $c$ where $y_c=1$ is
\[
\Pr(\vect y\mid\vect p)=\prod_k p_k^{y_k}=p_c
\]
(every factor with $y_k=0$ contributes $p_k^0=1$). Its negative log is
\[
-\log\Pr(\vect y\mid\vect p)=-\log p_c=-\sum_k y_k\log p_k,
\]
which is precisely the cross-entropy loss. So minimising cross-entropy over a dataset is exactly maximum-likelihood estimation of the model's class-probability parameters $\vect p=\operatorname{softmax}(\vect z)$.

\textbf{(b)} Computing $e^{z_k}$ directly overflows float64 once $z_k\gtrsim709$ --- long before the normalising sum $\sum_j e^{z_j}$ is even taken, so a naive implementation produces inf/nan well within the range of logits a real (even lightly-trained) network can produce. Subtracting $m=\max_j z_j$ from every logit before exponentiating fixes this while leaving $\vect p$ unchanged, because
\[
\operatorname{softmax}(\vect z-m)_k
=\frac{e^{z_k-m}}{\sum_i e^{z_i-m}}
=\frac{e^{z_k}e^{-m}}{\sum_i e^{z_i}e^{-m}}
=\frac{e^{z_k}}{\sum_i e^{z_i}}
=\operatorname{softmax}(\vect z)_k,
\]
i.e. the common factor $e^{-m}$ cancels between numerator and denominator. After the shift the largest exponent is $z_{\max}-m=0$, so $e^{z_k-m}\le1$ for every $k$ --- no term can overflow, and this is exactly the log-sum-exp trick:
\[
\log\sum_j e^{z_j}=m+\log\sum_j e^{z_j-m},
\]
used implicitly in \code{losses.py}'s \code{z_stable = z - z.max(axis=1, keepdims=True)} before taking \code{np.exp}.
\end{answerbox}

\qtitle{A5 --- Optimizers: momentum and Adam}{[5 pts]}
\begin{answerbox}
\textbf{(a)} SGD with momentum keeps a velocity $\vect v_t=\mu\vect v_{t-1}+\vect g_t$ and updates $\vect\theta_t=\vect\theta_{t-1}-\eta\vect v_t$ (matching \code{src/optimizers/momentum.py}). $\vect v_t$ is an exponentially-weighted running sum of past gradients: unrolling the recursion gives $\vect v_t=\sum_{i=1}^t\mu^{t-i}\vect g_i$. If consecutive gradients $\vect g_t$ point in a broadly consistent direction, their contributions to $\vect v_t$ add constructively, so $\vect v_t$ grows larger than any single $\vect g_t$ --- the step in that direction accelerates. If the gradient sign oscillates from step to step (e.g. across a narrow ravine), the terms partially cancel in the sum and the step shrinks --- this is what damps the oscillation.

\textbf{(b)} Adam tracks two exponential moving averages per parameter: a first moment
\[
\vect m_t=\beta_1\vect m_{t-1}+(1-\beta_1)\vect g_t
\]
(a smoothed direction estimate, playing the same role as momentum's $\vect v_t$) and a second moment
\[
\vect v_t=\beta_2\vect v_{t-1}+(1-\beta_2)\vect g_t^2
\]
(a smoothed estimate of the gradient's squared magnitude, used to adapt the step size per parameter: a parameter with large recent gradients gets a smaller effective step, and a parameter with small/sparse gradients gets a relatively larger one). Both start at $\vect m_0=\vect v_0=0$, which biases the early estimates toward zero --- e.g. at $t=1$, $\vect m_1=(1-\beta_1)\vect g_1$, only 10\% of $\vect g_1$ when $\beta_1=0.9$. The bias-corrected forms $\widehat{\vect m}_t=\vect m_t/(1-\beta_1^t)$, $\widehat{\vect v}_t=\vect v_t/(1-\beta_2^t)$ rescale the estimate back up to what a fully-converged running average would already show (this is proven in general in Bonus 3). The update is
\[
\vect\theta_t=\vect\theta_{t-1}-\eta\frac{\widehat{\vect m}_t}{\sqrt{\widehat{\vect v}_t}+\epsilon},
\]
matching \code{src/optimizers/adam.py} exactly, including $\epsilon$ in the denominator purely to avoid division by zero when a parameter's gradient has been consistently near-zero.
\end{answerbox}

\qtitle{A6 --- Regularization: weight decay \& dropout}{[3 pts]}
\begin{answerbox}
\textbf{(a)} With the penalised loss $\widetilde L=L+\frac{\lambda}{2}\lVert\vect W\rVert^2$, and since
\[
\frac{\partial}{\partial W_{ij}}\left(\frac{\lambda}{2}\sum_{ij}W_{ij}^2\right)=\lambda W_{ij},
\]
\[
\frac{\partial\widetilde L}{\partial\vect W}=\frac{\partial L}{\partial\vect W}+\lambda\vect W.
\]
Plugging this into an SGD step gives $\vect W\leftarrow\vect W-\eta(\partial L/\partial\vect W+\lambda\vect W)$ --- exactly the extra \code{+ self.weight_decay * p.data} term added to the raw gradient in every optimizer in \code{src/optimizers/} before the update, i.e. weight decay is implemented as $L_2$ regularisation added directly at the gradient level rather than as a separate penalty term computed in the loss.

\textbf{(b)} At train time each unit's output is kept with probability $1-p$ and, when kept, divided by $1-p$ (inverted dropout, matching \code{Dropout.forward}: \code{mask = (rng.random(...) < keep_prob) / keep_prob}). Its expectation is
\[
\mathbb E[\text{output}]=(1-p)\cdot\frac{x}{1-p}+p\cdot0=x,
\]
unchanged from the un-dropped activation $x$. Because the train-time expectation already equals the raw activation, eval time can simply turn dropout off and use the raw activation with no rescaling --- exactly what \code{Dropout.forward} does when \code{self.training} is False: it returns $x$ untouched.
\end{answerbox}

\sectitle{2}{Part B --- Build (60 pts)}

\qtitle{B.1 Implementation summary}{[maps to B1--B5]}
\begin{answerbox}
Tensor (\code{src/autograd/tensor.py}) gives a backward closure to every op it defines: \code{add}, \code{mul}, \code{matmul}, \code{sum}, \code{relu}, \code{log}, \code{exp} (plus \code{sqrt} and a reciprocal node so division stays a first-class graph op rather than a detached constant), and a \code{clip} primitive added during review, discussed below. \code{backward()} runs a DFS-based topological sort rooted at the loss, seeds \code{self.grad = 1}, then calls every node's backward in reverse topological order, so by the time a node's closure runs, every consumer of that node has already pushed its share of gradient into it (\code{Tensor.grad} accumulates with \code{+=}, which is what lets a tensor used twice in the graph correctly sum both contributions). Linear is \code{x @ self.W + self.b} with He-initialised $\vect W$ ($\operatorname{std}=\sqrt{2/\text{fan\_in}}$, chosen because it keeps activation variance roughly constant through a ReLU network, where half the units are zeroed on average at every layer). \code{tanh}/\code{sigmoid} are composed purely from \code{exp/add/mul/clip}, so their gradients come for free from the engine --- no hand-derived backward is written for them anywhere. \code{softmax_cross_entropy} computes the max-shifted softmax in plain NumPy and installs one fused backward that writes \code{(p-one_hot(y))/N} directly into \code{logits.grad} --- the one deliberate exception to ``build everything from Tensor ops'', justified by the A2(b) derivation above. Dropout multiplies by a random $\{0,1/(1-p)\}$ mask, itself a plain Tensor with no parents, so it participates in backprop as an ordinary elementwise gate; it is a no-op when \code{self.training} is False. MLP stacks Linear layers with \code{relu} (and, optionally, Dropout) between them, leaving the final layer as raw logits (softmax lives inside the loss, never inside the model). All three required optimizers plus the RMSProp bonus (\code{src/optimizers/}) share an abstract Optimizer base holding \code{params/lr/weight_decay}; Momentum keeps one velocity array per parameter, Adam keeps $m$ and $v$ arrays per parameter plus a step counter $t$ for bias correction; weight decay enters every optimizer identically, as \code{+ self.weight_decay * p.data} added to the raw gradient before the update rule is applied.

\textbf{A stability bug found and fixed during review.} The original \code{sigmoid}/\code{tanh} (built as \code{1/(1+exp(-x))} and from the raw exponential form of tanh) overflowed to \code{nan} for inputs with $|x|\gtrsim700$, since \code{exp} of a large-magnitude argument exceeds float64 range. The fix adds a \code{clip(min,max)} primitive to Tensor (backward: gradient passes through unchanged inside the bounds, zero outside) and uses it to bound the argument passed to \code{exp} inside sigmoid to $[-60,60]$ before exponentiating; tanh is then rewritten as $2\sigma(2x)-1$ so it inherits the fix automatically. Since $e^{60}\approx1.1\times10^{26}$ already saturates $\sigma$ to 0.0 or 1.0 at double precision, clipping there changes no value that mattered --- it only stops \code{exp} from being asked to produce a number bigger than float64 can hold. This exactly mirrors the max-subtraction stabilisation of A4(b): both fixes bound the argument of \code{exp} into a safe range without changing the mathematical result.
\end{answerbox}

\qtitle{B.8 Analysis}{[Part C readiness]}
\begin{answerbox}
\textbf{(i) Where did the trickiest component hide, and how did the gradient check find it?} The component that most deserved careful, non-finite-difference scrutiny was the fused softmax cross entropy backward, since it is hand-derived rather than composed automatically from existing Tensor ops --- every other component's gradient is a direct consequence of the generic engine, so a bug there would show up as a mismatched shape or an obviously wrong finite-difference error; a bug in a hand-written closure could instead be subtly wrong in a way that still points roughly downhill. \code{test_gradient_check_loss} covers it with the standard finite-difference check, but the test goes one step further and directly asserts \code{logits.grad} equals \code{(p - one_hot(y))/n} computed independently in NumPy --- so this one critical, hand-written component is checked two independent ways, not by finite differences alone. Separately, the review pass that produced this final version of the report did surface one real numerical bug (not a gradient-sign bug): \code{sigmoid}/\code{tanh} overflowed to \code{nan} on large-magnitude inputs, described in B.1. That bug would not have been caught by the gradient checks above at all, since gradient checks only probe well-behaved, moderate-magnitude inputs --- it needed a dedicated test with deliberately extreme inputs (\code{test_sigmoid_tanh_no_overflow_on_extreme_inputs}).

\textbf{(ii) What does the finite-difference check prove that a falling training loss does not?} A falling training loss only proves that some sequence of parameter updates reduced the loss on the batches seen --- that can happen even with a subtly wrong (but roughly loss-decreasing) gradient, or by luck on a small dataset. The finite-difference check instead verifies, component by component, that the analytic gradient produced by \code{backward()} numerically matches the true local slope of the loss surface at that point, independent of whether training happens to work downstream. It catches bugs --- a wrong sign, a wrong broadcasting axis, a missing chain-rule factor --- that could still let the loss go down but would make the model converge to the wrong place, or converge far slower than it should, without ever producing an obviously-broken training curve.
\end{answerbox}

\sectitle{3}{Part C --- Explain Your Code (15 pts, mandatory)}

\qtitle{C1 --- Trace one gradient backward through the Tensor graph}{[4 pts]}
\begin{answerbox}
Take the weight \code{lin1.W} of the first Linear layer. \code{backward()} starts at the scalar loss and seeds \code{self.grad = 1}. It first fires the \code{softmax_cross_entropy} closure, which writes \code{(p - one_hot) / n} directly into \code{logits.grad}. That flows into the second Linear's matmul and add closures (via \code{x @ self.W + self.b}), pushing a gradient into the hidden activation $\vect h$. Next comes the \code{relu} closure: it multiplies the incoming gradient elementwise by \code{(self.data > 0)}, so any hidden unit whose pre-activation $\vect z_1$ was $\le0$ has its gradient zeroed right there --- nothing further back through that unit changes as a result. Finally the first Linear's matmul closure fires, adding \code{self.data.T @ out.grad} (i.e. $\vect x^\top\cdot$ incoming gradient) into \code{lin1.W.grad}.
\end{answerbox}

\qtitle{C2 --- Contrast SGD, momentum, and Adam}{[4 pts]}
\begin{answerbox}
Given the identical \code{p.grad} stream: plain SGD (\code{sgd.py}) applies it directly, scaled only by one global \code{lr} --- every parameter takes the same relative step size regardless of its own gradient history. Momentum (\code{momentum.py}) accumulates a velocity per parameter, so it does not adapt the step size, but does adapt the effective direction/magnitude by smoothing across steps --- consistent gradients build speed. Adam (\code{adam.py}) is the only one of the three that adapts the step per-parameter: it divides by $\sqrt{\widehat{\vect v}_t}$, so parameters with large recent gradients get smaller effective steps and vice versa. In \code{optimizer_comparison.png} this shows up as SGD+Momentum and Adam both climbing faster than plain SGD in the first $\sim10$ epochs; Adam's per-parameter scaling is what lets it make quick early progress even though its base \code{lr} ($10^{-3}$) is far smaller than SGD's (0.1) or Momentum's (0.05).
\end{answerbox}

\qtitle{C3 --- Why does Dropout behave differently at train vs. eval?}{[4 pts]}
\begin{answerbox}
At train time, Dropout needs to inject randomness (zeroing random units via the mask \code{(rng.random(x.data.shape) < keep_prob) / keep_prob}) so the network cannot rely on any single unit and is forced to learn redundant, more robust features. At eval time we want one deterministic prediction, so \code{Dropout.forward} skips the mask entirely and returns $x$ unchanged when \code{self.training} is False. If the mode switch is forgotten, eval-time predictions become randomly noisy and non-reproducible across calls. If the \code{/keep_prob} rescaling is dropped at train time (or applied inconsistently), the expected magnitude of activations at train time no longer matches what the next layer sees at eval time --- train-time activations become systematically too small by a factor of $1-p$ (see A6(b)) --- which biases every downstream computation, including the loss and the gradients used to update the downstream weights.
\end{answerbox}

\qtitle{C4 --- Point to the line implementing the Adam bias correction}{[3 pts]}
\begin{answerbox}
From \code{src/optimizers/adam.py}:
\begin{codebox}
\code{m_hat = m / (1 - self.beta1 ** self.t)}\par
\code{v_hat = v / (1 - self.beta2 ** self.t)}
\end{codebox}
Without this, $m$ and $v$ start at zero and are updated by a weighted average that is still mostly zero for the first few steps --- e.g. at $t=1$, $m_1=(1-\beta_1)g_1$, only 10\% of $g_1$ for $\beta_1=0.9$. Dividing by $1-\beta_1^t$ (respectively $1-\beta_2^t$) rescales these early, artificially-small moment estimates back up to what they would be if the running average had already converged, so the very first parameter updates are not unnecessarily tiny relative to the true gradient signal.
\end{answerbox}

\qtitle{Bonus 3 --- Prove one gradient relied on}{[+3]}
\begin{answerbox}
Adam's bias-corrected $\widehat m_t=m_t/(1-\beta_1^t)$ is an unbiased estimate of $\mathbb E[g_t]$. Assume gradients are approximately stationary, $\mathbb E[g_i]\approx\mathbb E[g_t]=:g$ for all $i\le t$. Unrolling the recursion $m_t=\beta_1m_{t-1}+(1-\beta_1)g_t$ with $m_0=0$ gives $m_t=(1-\beta_1)\sum_{i=1}^t\beta_1^{t-i}g_i$. Taking expectations,
\[
\mathbb E[m_t]=(1-\beta_1)\sum_{i=1}^t\beta_1^{t-i}\mathbb E[g_i]
\approx g(1-\beta_1)\sum_{i=1}^t\beta_1^{t-i}
=g(1-\beta_1)\frac{1-\beta_1^t}{1-\beta_1}
=g(1-\beta_1^t),
\]
using the finite geometric series $\sum_{i=1}^t\beta_1^{t-i}=\sum_{k=0}^{t-1}\beta_1^k=(1-\beta_1^t)/(1-\beta_1)$. So $\mathbb E[m_t]=(1-\beta_1^t)g$ --- biased toward zero by the factor $(1-\beta_1^t)$, which is exactly why a raw (uncorrected) $m_t$ would produce artificially small early updates. Dividing through,
\[
\mathbb E[\widehat m_t]=\frac{\mathbb E[m_t]}{1-\beta_1^t}=g=\mathbb E[g_t],
\]
exactly unbiased. The identical argument, substituting $\beta_2$ and $g_t^2$ for $\beta_1$ and $g_t$, shows $\widehat v_t$ is an unbiased estimate of $\mathbb E[g_t^2]$.
\end{answerbox}
